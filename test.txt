# 1. GET — завантаження картинки з http.cat та збереження в кеш (очікується 200 OK, X-Cache: MISS)
curl -s -o /dev/null -D - http://127.0.0.1:3000/200

# 2. GET — повторний запит того ж зображення (очікується 200 OK, X-Cache: HIT)
curl -s -o /dev/null -D - http://127.0.0.1:3000/200

# 3. GET — завантаження іншого HTTP-статусу (наприклад, 418 I'm a teapot)
curl -s -o /dev/null -D - http://127.0.0.1:3000/418

# 4. GET — запит неіснуючого HTTP-коду (очікується 404 Not Found)
curl -i http://127.0.0.1:3000/999

# 5. PUT — ручне завантаження або заміна картинки в кеші (очікується 201 Created)
curl -i -X PUT --data-binary @./cache/200.jpg http://127.0.0.1:3000/777

# 6. GET — перевірка читання завантаженої через PUT картинки з кешу (очікується 200 OK, X-Cache: HIT)
curl -s -o /dev/null -D - http://127.0.0.1:3000/777

# 7. DELETE — видалення картинки з кешу (очікується 200 OK)
curl -i -X DELETE http://127.0.0.1:3000/777

# 8. DELETE — повторне видалення вже відсутнього файлу (очікується 404 Not Found)
curl -i -X DELETE http://127.0.0.1:3000/777

# 9. POST — перевірка непідтримуваного HTTP-методу (очікується 405 Method Not Allowed)
curl -i -X POST http://127.0.0.1:3000/200
